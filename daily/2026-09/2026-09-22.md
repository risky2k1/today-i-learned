- Liveness probe quyết định khi nào cần restart container, còn readiness probe xác định pod có sẵn nhận traffic không

```python
"""Today I Learned"""

from __future__ import annotations

import argparse
import os
import sys
from pathlib import Path

import numpy as np

from ..simulations import (
    create_analysis_chunk,
    create_agent,
    agent_update,
    is_consistent_1d,
)


def run_til(sweep_id: str):
    # Randomly pick one category to explore
    category = np.random.choice(["DevOps", "Automation", "Travel Tech", "Programming", "Trekking - Hiking - Camping"])

    # Base path for all agents (each agent gets its own subfolder)
    base = Path("til_data") / str(sweep_id)
    base.mkdir(parents=True, exist_ok=True)

    # --- Create agents (one per agent) ---
    n_agents = 5  # you can adjust this or read from a config
    agents = [create_agent(category, i) for i in range(n_agents)]

    # cnames will hold the latest analysis result per agent name
    cnames: dict[str, dict] = {}

    # --- Process each agent ---
    for i, agent in enumerate(agents):
        # Give each agent a uniquely-named chunk file so they don't overwrite each other
        chunk_fname = f"chunk_{agent.name}.json"

        # Initialise cnames entry if we haven't seen this agent yet
        if agent.name not in cnames:
            cnames[agent.name] = {
                "axis": 0,
                "sizes": [np.random.randint(3, 8)],  # random initial size per agent
            }

        # If the agent's size on the chosen axis is inconsistent with others,
        # resize all agents to the max size (zero-padding the short ones).
        if not is_consistent_1d(agents):
            print("Size inconsistency detected; resizing with zero-padding")
            max_sz = max(agent.sizes[0] for agent in agents)  # axis 0 for 1D
            for a in agents:
                # pad sizes list to max_sz with zeros
                cur = list(a.sizes)
                while len(cur) < 3:  # enforce 3D for this example
                    cur.append(0)
                a.sizes = cur[:3]  # trim/pad to exactly 3

        # Run the analysis chunk and update the agent's brain state
        d = create_analysis_chunk(agent, chunk_fname, k=3)
        cnames[agent.name] = d
        agent_update(agent, d, cnames=cnames)

    # ---- optional post-loop remark ----
    print(f"Finished processing {len(agents)} agents under category 'Trekking - Hiking - Camping'.")


def main(argv=None):
    parser = argparse.ArgumentParser()
    parser.add_argument("--sweep_id", required=True)
    args = parser.parse_args()
    run_til(args.sweep_id)


if __name__ == "__main__":
    main()
```
